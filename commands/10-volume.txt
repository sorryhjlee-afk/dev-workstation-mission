#  [사용법 - 꼭 bash 터미널에서 하세요]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널]  (단축키 Ctrl + `)
#   3. 터미널 패널 오른쪽 위 [v] 를 눌러 'Git Bash' 를 고릅니다.
#      프롬프트가  사용자@PC MINGW64 ~ $  처럼 '$' 로 끝나면 bash 입니다.
#      'PS C:\...>' 로 보이면 PowerShell 이니 반드시 Git Bash 로 바꾸세요.
#   4. 아래 명령을 '한 줄씩' 복사해서 붙여넣고 Enter 를 누릅니다.
#   5. # 으로 시작하는 줄은 설명이라 같이 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   컨테이너는 '지우면 안의 내용도 같이 사라집니다'.
#   그래서 지켜야 할 자료는 컨테이너 바깥의 볼륨(보관함)에 넣습니다.
#   정말 남는지 '컨테이너를 삭제해서' 직접 확인합니다.
# ============================================================================

# (0) ★ 윈도우 Git Bash 전용 준비 한 줄 (맥/리눅스에서는 필요 없습니다)
#     Git Bash 는 /usr/share/... 처럼 / 로 시작하는 글자를 윈도우 경로로
#     자동으로 바꿔 버립니다. 그러면 docker 명령이 엉뚱한 곳을 가리킵니다.
#     터미널을 새로 열 때마다 이 한 줄을 먼저 실행하세요.
export MSYS_NO_PATHCONV=1

# (1) 볼륨 만들기
docker volume create my-data

# (2) 볼륨 목록 확인
docker volume ls --filter name=my-data

# (3) 볼륨을 연결해서 컨테이너 실행 (8083)
docker run -d -p 8083:80 --name vol-8083 -v my-data:/usr/share/nginx/html/data my-web-server:1.0

# (4) 볼륨 안에 메모 남기기
docker exec vol-8083 sh -c 'echo hello-volume > /usr/share/nginx/html/data/memo.txt'

# (5) 잘 들어갔는지 확인 (터미널 + 브라우저)
docker exec vol-8083 cat /usr/share/nginx/html/data/memo.txt
curl -s http://localhost:8083/data/memo.txt
cmd.exe //c start "" "http://localhost:8083/data/memo.txt"

# (6) ★ 컨테이너를 통째로 삭제합니다 (무서워도 괜찮아요)
docker rm -f vol-8083

# (7) 정말 사라졌는지 확인 (목록이 비어 있어야 정상)
docker ps -a --filter name=vol-8083

# (8) 볼륨은 남아 있습니다
docker volume ls --filter name=my-data

# (9) 새 컨테이너에 같은 볼륨을 연결
docker run -d -p 8083:80 --name vol-8083-new -v my-data:/usr/share/nginx/html/data my-web-server:1.0

# (10) 메모가 그대로 있는지 확인
docker exec vol-8083-new cat /usr/share/nginx/html/data/memo.txt
curl -s http://localhost:8083/data/memo.txt

# (11) 볼륨이 어디에 저장되는지 구경
docker volume inspect my-data
