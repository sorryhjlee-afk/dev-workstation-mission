#  [사용법 - 꼭 bash 터미널에서 하세요]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널]  (단축키 Ctrl + `)
#   3. 터미널 패널 오른쪽 위 [v] 를 눌러 'Git Bash' 를 고릅니다.
#      프롬프트가  사용자@PC MINGW64 ~ $  처럼 '$' 로 끝나면 bash 입니다.
#      'PS C:\...>' 로 보이면 PowerShell 이니 반드시 Git Bash 로 바꾸세요.
#   4. 아래 명령을 '한 줄씩' 복사해서 붙여넣고 Enter 를 누릅니다.
#   5. # 으로 시작하는 줄은 설명이라 같이 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   바인드 마운트 = 내 컴퓨터 폴더를 컨테이너 안에 그대로 '겹쳐 놓기'
#   컨테이너가 내 폴더를 실시간으로 보기 때문에,
#   파일을 고치면 다시 빌드하지 않아도 새로고침만으로 바뀝니다.
# ============================================================================

# (0) ★ 윈도우 Git Bash 전용 준비 한 줄 (맥/리눅스에서는 필요 없습니다)
#     Git Bash 는 /usr/share/... 처럼 / 로 시작하는 글자를 윈도우 경로로
#     자동으로 바꿔 버립니다. 그러면 docker 명령이 엉뚱한 곳을 가리킵니다.
#     터미널을 새로 열 때마다 이 한 줄을 먼저 실행하세요.
export MSYS_NO_PATHCONV=1

# (1) 내 app 폴더를 컨테이너의 웹 폴더에 겹쳐서 실행 (8082)
#     $(pwd) 는 '지금 있는 폴더' 를 뜻합니다. 어느 PC 에서도 그대로 동작합니다.
docker run -d -p 8082:80 --name bind-8082 -v "$(pwd)/app:/usr/share/nginx/html" my-web-server:1.0

# (2) 지금 화면 내용 확인 (변경 전)
curl -s http://localhost:8082/ | grep "<h1>"

# (3) VS Code 에서 app/index.html 을 열어 아래 두 군데를 고치고 저장(Ctrl+S)
#       <h1>Hello Docker!</h1>   ->   <h1>안녕하세요, 도커!</h1>
#       version 1.0              ->   version 2.0 (수정됨)
#     * 저장할 때 인코딩은 UTF-8 이어야 한글이 안 깨집니다 (VS Code 는 기본이 UTF-8)
#     * 터미널에서 바로 바꾸고 싶다면 (sed 는 글자를 찾아 바꿔 주는 도구입니다)
#         sed -i 's|<h1>Hello Docker!</h1>|<h1>안녕하세요, 도커!</h1>|' app/index.html
#         sed -i 's|version 1.0|version 2.0 (수정됨)|' app/index.html

# (4) 다시 확인 (변경 후) - 빌드를 다시 하지 않았는데 바뀝니다
curl -s http://localhost:8082/ | grep "<h1>"

# (5) 같은 순간 8080 은? (COPY 로 구워 넣은 컨테이너라 그대로입니다)
curl -s http://localhost:8080/ | grep "<h1>"

# (6) 브라우저로도 비교해 보세요
cmd.exe //c start "" "http://localhost:8082/"
cmd.exe //c start "" "http://localhost:8080/"

# (7) 실습이 끝나면 index.html 을 원래대로 되돌려 놓으세요.
#       sed -i 's|<h1>안녕하세요, 도커!</h1>|<h1>Hello Docker!</h1>|' app/index.html
#       sed -i 's|version 2.0 (수정됨)|version 1.0|' app/index.html

# [성공하면?]
#   8082 만 바뀌고 8080 은 그대로면 성공입니다.
#   같은 이미지인데 결과가 다른 이유:
#     8082 = 내 폴더를 실시간으로 봄 (창문)
#     8080 = 빌드할 때 찍어 둔 사진을 봄 (복사본)
