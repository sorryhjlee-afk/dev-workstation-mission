# ============================================================================
#  STEP 02. 터미널 기본 조작 (bash)
# ============================================================================
#  [사용법 - 꼭 bash 터미널에서 하세요]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널]  (단축키 Ctrl + `)
#   3. 터미널 패널 오른쪽 위 [v] 를 눌러 'Git Bash' 를 고릅니다.
#      프롬프트가  사용자@PC MINGW64 ~ $  처럼 '$' 로 끝나면 bash 입니다.
#      'PS C:\...>' 로 보이면 PowerShell 이니 반드시 Git Bash 로 바꾸세요.
#   4. 아래 명령을 '한 줄씩' 복사해서 붙여넣고 Enter 를 누릅니다.
#   5. # 으로 시작하는 줄은 설명이라 같이 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   폴더를 만들고, 들어가고, 파일을 만들고, 복사·이름변경·이동·삭제하고,
#   내용을 읽어 봅니다. 과제의 "터미널 조작 로그" 항목입니다.
#
#   * 절대 경로 : /d/22.코드아카데미/dev-workstation-mission 처럼 맨 앞 / 부터 쓰는 '전체 주소'
#     (bash 에서 윈도우 D: 드라이브는 /d 로 보입니다)
#   * 상대 경로 : ./app 처럼 '지금 있는 자리' 기준의 짧은 주소
#     (.. 는 한 칸 위 폴더, . 은 지금 폴더, ~ 는 내 홈 폴더)
# ============================================================================

# (1) 나는 지금 어디에 있나?  (절대 경로로 나옵니다)
pwd

# (2) 이 폴더에 뭐가 있나?  -a 를 붙이면 숨김 파일(.으로 시작)도 보입니다
ls -la

# (3) 실습용 폴더 만들기 (-p 는 '없으면 만들고 있으면 넘어가라')
mkdir -p .lab-terminal

# (4) 그 폴더로 이동 (상대 경로 사용)
cd ./.lab-terminal
pwd

# (5) 빈 파일 만들기
touch empty.txt
ls -l empty.txt

# (6) 내용이 있는 파일 만들고 읽어 보기
echo "hello docker" > hello.txt
cat hello.txt

# (7) 복사하기
cp hello.txt copy.txt
ls

# (8) 이름 바꾸기 (mv 는 '옮기기' 인데 같은 자리면 '이름 바꾸기' 가 됩니다)
mv copy.txt renamed.txt
ls

# (9) 다른 폴더로 옮기기
mkdir -p sub
mv renamed.txt ./sub/
ls -R

# (10) 삭제하기
rm ./sub/renamed.txt
ls -R

# (11) 한 칸 위로 나와서 실습 폴더 통째로 지우기
cd ..
rm -rf .lab-terminal
pwd

# [성공하면?]
#   (9) 에서 sub/renamed.txt 가 보이고, (10) 뒤에는 사라져 있으면 OK.
#
# [조심할 것]
#   rm -rf 는 '묻지 않고 통째로 삭제' 입니다. 지울 폴더 이름을 꼭 두 번 확인하세요.
