# ============================================================================
#  STEP 02. 터미널 기본 조작
# ============================================================================
#  [사용법]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널] 을 누릅니다.  (단축키 Ctrl + `)
#   3. 아래 명령을 '한 줄씩' 복사해서 터미널에 붙여넣고 Enter 를 누릅니다.
#   4. # 으로 시작하는 줄은 설명이라 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   폴더를 만들고, 들어가고, 파일을 만들고, 복사·이름변경·이동·삭제하고,
#   내용을 읽어 봅니다. 과제의 "터미널 조작 로그" 항목입니다.
#
#   * 절대 경로 : C:\study\my-web-server 처럼 드라이브부터 쓰는 '전체 주소'
#   * 상대 경로 : .\app 처럼 '지금 있는 자리'를 기준으로 한 짧은 주소
#     (.. 는 한 칸 위 폴더, . 은 지금 폴더)
# ============================================================================

# (1) 나는 지금 어디에 있나?  (절대 경로로 나옵니다)
Get-Location

# (2) 이 폴더에 뭐가 있나?  -Force 를 붙이면 숨김 파일(.으로 시작)도 보입니다
Get-ChildItem -Force

# (3) 실습용 폴더 만들기
New-Item -ItemType Directory -Path .lab-terminal -Force

# (4) 그 폴더로 이동 (상대 경로 사용)
Set-Location .\.lab-terminal
Get-Location

# (5) 빈 파일 만들기
New-Item -ItemType File -Name empty.txt

# (6) 내용이 있는 파일 만들고 읽어 보기
Set-Content -Path hello.txt -Value "hello docker" -Encoding UTF8
Get-Content hello.txt

# (7) 복사하기
Copy-Item hello.txt copy.txt
Get-ChildItem -Name

# (8) 이름 바꾸기
Rename-Item copy.txt renamed.txt
Get-ChildItem -Name

# (9) 다른 폴더로 옮기기
New-Item -ItemType Directory -Name sub
Move-Item renamed.txt .\sub\
Get-ChildItem -Recurse -Name

# (10) 삭제하기
Remove-Item .\sub\renamed.txt
Get-ChildItem -Recurse -Name

# (11) 한 칸 위로 나와서 실습 폴더 통째로 지우기
Set-Location ..
Remove-Item .lab-terminal -Recurse -Force

# [성공하면?]
#   (9) 에서 sub\renamed.txt 가 보이고, (10) 뒤에는 사라져 있으면 OK.
