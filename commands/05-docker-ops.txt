# ============================================================================
#  [사용법 - 꼭 bash 터미널에서 하세요]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널]  (단축키 Ctrl + `)
#   3. 터미널 패널 오른쪽 위 [v] 를 눌러 'Git Bash' 를 고릅니다.
#      프롬프트가  사용자@PC MINGW64 ~ $  처럼 '$' 로 끝나면 bash 입니다.
#      'PS C:\...>' 로 보이면 PowerShell 이니 반드시 Git Bash 로 바꾸세요.
#   4. 아래 명령을 '한 줄씩' 복사해서 붙여넣고 Enter 를 누릅니다.
#   5. # 으로 시작하는 줄은 설명이라 같이 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   이미지를 받고, 목록을 보고, 컨테이너를 켜고 끄고, 로그와 자원 사용량을 봅니다.
#
#   * 이미지(image)       = 요리 '레시피'  (아직 안 움직임)
#   * 컨테이너(container) = 레시피로 만든 '진짜 요리' (움직임)
# ============================================================================

# (1) 이미지 내려받기
docker pull nginx:latest

# (2) 내 컴퓨터에 있는 이미지 목록
docker images

# (3) 컨테이너 실행 (-d 는 뒤에서 조용히 실행하라는 뜻)
docker run -d --name ops-demo nginx:latest

# (4) 실행 중인 컨테이너 목록
docker ps

# (5) 잠깐 멈추기
docker stop ops-demo

# (6) 멈춘 것은 docker ps 에는 안 보입니다
docker ps

# (7) -a 를 붙이면 멈춘 것까지 전부 보입니다
docker ps -a --filter name=ops-demo

# (8) 다시 켜기
docker start ops-demo

# (9) 컨테이너가 남긴 일기장(로그) 보기
docker logs ops-demo

# (10) 자원을 얼마나 쓰는지 (--no-stream 은 한 번만 찍고 끝)
docker stats --no-stream ops-demo

# (11) 실습용 컨테이너 치우기
docker rm -f ops-demo

# [성공하면?]
#   (5) 뒤 STATUS 가 Exited (0) 으로 바뀌고 PORTS 가 비어 있으면 정상입니다.
#   stop 은 '잠깐 재우기', rm 은 '버리기' 입니다. 헷갈리지 마세요.
