#  [사용법 - 꼭 bash 터미널에서 하세요]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널]  (단축키 Ctrl + `)
#   3. 터미널 패널 오른쪽 위 [v] 를 눌러 'Git Bash' 를 고릅니다.
#      프롬프트가  사용자@PC MINGW64 ~ $  처럼 '$' 로 끝나면 bash 입니다.
#      'PS C:\...>' 로 보이면 PowerShell 이니 반드시 Git Bash 로 바꾸세요.
#   4. 아래 명령을 '한 줄씩' 복사해서 붙여넣고 Enter 를 누릅니다.
#   5. # 으로 시작하는 줄은 설명이라 같이 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   -p 호스트포트:컨테이너포트  =  내 컴퓨터 포트와 컨테이너 포트를 잇는 '통로'
#
#   왜 필요할까요? 컨테이너 안 nginx 는 언제나 80번을 씁니다.
#   그런데 내 컴퓨터의 80번은 하나뿐이라 여러 개를 동시에 못 띄웁니다.
#   그래서 바깥 번호만 8080, 8081 처럼 다르게 주는 것입니다.
# ============================================================================

# (0) ★ 윈도우 Git Bash 전용 준비 한 줄 (맥/리눅스에서는 필요 없습니다)
#     Git Bash 는 /usr/share/... 처럼 / 로 시작하는 글자를 윈도우 경로로
#     자동으로 바꿔 버립니다. 그러면 docker 명령이 엉뚱한 곳을 가리킵니다.
#     터미널을 새로 열 때마다 이 한 줄을 먼저 실행하세요.
export MSYS_NO_PATHCONV=1

# (1) 첫 번째 : 내 컴퓨터 8080 -> 컨테이너 80
docker run -d -p 8080:80 --name web-8080 my-web-server:1.0

# (2) 두 번째 : 같은 이미지를 8081 로 한 번 더
docker run -d -p 8081:80 --name web-8081 my-web-server:1.0

# (3) 둘 다 떠 있는지 확인
docker ps

# (4) 터미널로 접속 테스트 (응답 헤더 보기)
curl -s -D - -o /dev/null http://localhost:8080/
curl -s -D - -o /dev/null http://localhost:8081/

# (5) 본문이 잘 오는지도 확인
curl -s http://localhost:8080/ | grep "<h1>"

# (6) 내가 바꾼 환경 변수 확인
docker exec web-8080 printenv APP_ENV

# (7) 브라우저로 열기  -> 주소창이 보이게 캡처하세요 (Windows 키 + Shift + S)
cmd.exe //c start "" "http://localhost:8080/"
cmd.exe //c start "" "http://localhost:8081/"
#     맥이면  open http://localhost:8080/
#     리눅스면 xdg-open http://localhost:8080/
#     귀찮으면 그냥 브라우저 주소창에 직접 쳐도 됩니다.

# [성공하면?]
#   HTTP/1.1 200 OK 와 X-Mission: codyssey-m1 헤더가 보이면 성공입니다.
#   X-Mission 은 내가 교체한 nginx 설정이 적용됐다는 증거입니다.
#
# [실패하면?]
#   "port is already allocated"  -> 그 번호를 이미 누가 쓰는 중. docker ps 로 확인 후 다른 번호 사용
#   "name is already in use"     -> docker rm -f web-8080 으로 지우고 다시 실행
