# (1) 지금 무엇이 떠 있는지 확인
docker ps -a

# (2) 이번 실습에서 만든 컨테이너만 삭제
docker rm -f web-8080 web-8081 bind-8082 vol-8083-new

# (3) 볼륨 삭제 (안에 있던 데이터도 함께 사라집니다)
docker volume rm my-data

# (4) 이미지 삭제
docker image rm my-web-server:1.0

# (5) 보너스(Compose)도 했다면 같이 정리
docker compose down

# (6) 보너스 볼륨까지 지우기 (* 안의 데이터가 사라집니다)
# docker compose down -v
# docker image rm my-web-server:compose

# (7) 정리 후 확인
docker ps -a
docker images
