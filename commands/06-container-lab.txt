# ============================================================================
#  STEP 06. 컨테이너 실행 실습 (hello-world / ubuntu / exec vs attach)
# ============================================================================
#  [사용법]
#   1. VS Code 에서 이 저장소 폴더를 엽니다.  (파일 - 폴더 열기)
#   2. 메뉴 [터미널] - [새 터미널] 을 누릅니다.  (단축키 Ctrl + `)
#   3. 아래 명령을 '한 줄씩' 복사해서 터미널에 붙여넣고 Enter 를 누릅니다.
#   4. # 으로 시작하는 줄은 설명이라 복사해도 실행되지 않습니다.
#
#  [무엇을 하나요?]
#   컨테이너 안으로 직접 들어가 명령을 내려 보고,
#   exec 와 attach 가 어떻게 다른지 눈으로 확인합니다.
# ============================================================================

# (1) 가장 작은 컨테이너로 동작 확인 (--rm 은 끝나면 자동 삭제)
docker run --rm hello-world

# (2) ubuntu 컨테이너를 뒤에서 켜 두기 (-dit : 뒤에서 + 터미널 유지)
docker run -dit --name lab-ubuntu ubuntu:24.04 bash

# (3) 컨테이너 안의 폴더 목록 보기
docker exec lab-ubuntu ls /

# (4) 컨테이너 안에서 글자 출력하고 리눅스 종류 확인
docker exec lab-ubuntu bash -c "echo hello-from-container; head -2 /etc/os-release"

# (5) 컨테이너 안으로 직접 들어가 보기  (나올 때는 exit 라고 치세요)
docker exec -it lab-ubuntu bash

# (6) 안에서 돌고 있는 프로세스 목록 - PID 1 이 '주 프로세스' 입니다
docker exec lab-ubuntu ps -ef

# (7) attach 를 실험하기 위한 컨테이너 (3초 뒤 할 일이 끝남)
docker run -d --name lab-attach ubuntu:24.04 sh -c "sleep 3; echo main-process-finished"

# (8) attach 로 주 프로세스 화면에 붙기 - 주 프로세스가 끝나면 같이 끝납니다
docker attach --no-stdin lab-attach

# (9) 주 프로세스가 끝났으니 컨테이너는 Exited
docker ps -a --filter name=lab-attach --format "{{.Names}} / {{.Status}}"

# (10) 반대로 exec 만 쓴 컨테이너는 아직 살아 있습니다
docker ps -a --filter name=lab-ubuntu --format "{{.Names}} / {{.Status}}"

# (11) 정리
docker rm -f lab-ubuntu lab-attach

# [exec 와 attach 차이 정리]
#   exec   = 컨테이너 안에 '새 프로세스를 하나 더' 만든다. 빠져나와도 컨테이너는 산다.
#   attach = 컨테이너의 '1번(주) 프로세스' 화면에 붙는다. 그게 끝나면 컨테이너도 멈춘다.
